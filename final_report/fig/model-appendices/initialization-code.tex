\vspace{0pt}\textbf{Shared lifecycle: ModelObject.initialize()}\par\smallskip
\begin{lstlisting}[style=report-python,basicstyle=\ttfamily\fontsize{8.5}{10.88}\selectfont]
def initialize(self):
    if not self._initialized:
        self.create_defaults()
        self._initialized = True
    if self.mission is not None:
        for item in self.walk():
            if not item._requirements_initialized:
                item.create_requirements()
                item._requirements_initialized = True
    return self
\end{lstlisting}

\vspace{0pt}\textbf{Automatic component addition: Subsystem.create\_defaults()}\par\smallskip
\begin{lstlisting}[style=report-python,basicstyle=\ttfamily\fontsize{8.5}{10.88}\selectfont]
def create_defaults(self):
    for cls in self.default_components:
        if not any(child.name == cls.__name__ for child in self.children):
            self.add(cls(cls.__name__))
\end{lstlisting}
